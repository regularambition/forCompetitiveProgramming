/**
 * 内部的にTrieを利用することでAho-Corasick法を実現するクラス
 * [verification]
 * https://judge.yosupo.jp/submission/412640
 */
class AhoCorasick {
  static constexpr ll SIGMA = 26;
  static constexpr ll NONEXISTENT_NODE_INDEX = -1;

  struct Node {
    ll parent;
    /**
     * dest[i] := 自身が表す文字列の末尾に対し
     * 利用可能な文字のうち辞書順でi番目に来るものを連結して得られる
     * 文字列の接尾辞であってTrieの中に存在する
     * という条件を満たす最長のものに対応するノード番号
     */
    array<ll, SIGMA> dest;
    /**
     * 自身が表す文字列の真の接尾辞であってTrieの中に存在するという
     * 条件を満たす最長のものに対応するノード番号
     */
    ll failure_link;

    // 実際に存在する文字列の末尾に対応するノードである場合に限りtrue
    bool is_terminal;

    /**
     * failure linkに従って遷移を一度以上行った結果として見つかる
     * 実際に存在する文字列の末尾に対応するノード番号
     * ただしそのような真の接尾辞が存在しない場合は根のノード番号
     */
    ll output_link;

    Node() : parent(NONEXISTENT_NODE_INDEX), failure_link(0), is_terminal(false), output_link(0) {
      fill(ALL(dest), NONEXISTENT_NODE_INDEX);
    }
  };

  vector<Node> nodes;

public:
  AhoCorasick() {
    // 根に対応するノードが一つだけ存在する状態で初期化
    nodes.emplace_back();
  }

  /**
   * Trieに対して文字列sを追加し
   * sの末尾に対応するノード番号を返す
   * 時間計算量はO(|s|)
   */
  ll add_string(const string &s) {
    ll v = 0;
    for (const char &ch : s) {
      ll u = nodes[v].dest[ch - 'a'];

      if (u == NONEXISTENT_NODE_INDEX) {
        const ll tmp = nodes.size();
        u = nodes[v].dest[ch - 'a'] = tmp;
        nodes.emplace_back();
      }

      nodes[u].parent = v;
      v = u;
    }

    nodes[v].is_terminal = true;

    return v;
  }

  /**
   * Trie内部においてfailure-linkなどを計算する
   * 時間計算量はTrieが持つノード数をNとしてO(N)
   */
  void build() {
    queue<ll> que;
    rep(i, SIGMA) {
      ll u = nodes[0].dest[i];

      if (u == NONEXISTENT_NODE_INDEX) {
        // Trieに登録された接尾辞は自明に空文字列のみ
        nodes[0].dest[i] = 0;
      } else {
        // Trieに登録された真の接尾辞は自明に空文字列のみ
        nodes[u].failure_link = 0;
        que.push(u);
      }
    }

    while (!que.empty()) {
      ll v = que.front();
      que.pop();

      const ll f = nodes[v].failure_link;
      nodes[v].output_link = nodes[f].is_terminal ? f : nodes[f].output_link;

      rep(i, SIGMA) {
        const ll u = nodes[v].dest[i];
        if (u == NONEXISTENT_NODE_INDEX) {
          nodes[v].dest[i] = nodes[f].dest[i];
        } else {
          nodes[u].failure_link = nodes[f].dest[i];
          que.push(u);
        }
      }
    }
  }

  // Trieが持つノードの個数を返す
  ll size() {
    return nodes.size();
  }

  // Trie内部におけるノード番号を受け取りその直接の親にあたるノード番号を返す
  ll parent_of_node(ll v) {
    return nodes[v].parent;
  }

  /**
   * Trie内部におけるノード番号を受け取り
   * failure-linkを辿った先にあたるノード番号を返す
   */
  ll failure_link_of_node(ll v) {
    return nodes[v].failure_link;
  }

  /**
   * Trie内部におけるノード番号を受け取り
   * 自身が表す文字列の末尾に対して
   * 文字chを連結させて得られる文字列に対応するノード番号を返す
   */
  ll to_direct_child(ll v, char ch) {
    return nodes[v].dest[ch - 'a'];
  }

  /**
   * Trie内部におけるノード番号を受け取り
   * output-linkを辿った先にあたるノード番号を返す
   */
  ll output_link_of_node(ll v) {
    return nodes[v].output_link;
  }

  /**
   * Trie内部におけるノード番号を受け取り
   * それを末尾とする文字列が一つ以上存在する場合はtrue
   * そうでなければfalseを返す
   */
  bool is_terminal_node(ll v) {
    return nodes[v].is_terminal;
  }
};
