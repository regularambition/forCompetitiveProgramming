/**
 * 長さがnの文字列sを受け取って
 * Z[i] := sとs[i,n)の最長共通接頭辞の長さ
 * と定義される整数列Zを返す
 * 時間計算量はO(n)
 */
vector<ll> z_algorithm(const string &s) {
  const ll n = s.length();

  // res[i]の定義
  // sという文字列全体と
  // s[i,n)即ちs[i]を先頭文字とするsの接尾辞にあたる文字列の
  // 最長共通接頭辞の長さ
  vector<ll> res(n, 0);
  res[0] = n;

  // 過去の探索によって
  // s[l,r) = s[0,r-l)
  // であると判明していることを表す整数l,rを用意
  ll l = 0, r = 0;

  rep_closed(i, 1, n - 1) {
    // l <= i < rなるiについて
    // sの先頭から考えた場合の位置に対応するのはl-i
    if (i < r) {
      res[i] = min(res[i - l], r - i);
    }

    // rは最大でもnにしかなり得ず
    // 比較が成功する場合は必ずrが大きくなるため
    // 全てのiに対するwhileブロック内部に入る回数の合計値が
    // nを超えることはない
    // 従って二重ループではあるがO(n^2)にはならない
    while (i + res[i] < n && s[res[i]] == s[i + res[i]]) {
      ++res[i];
    }

    if (i + res[i] > r) {
      l = i;
      r = i + res[i];
    }
  }

  return res;
}
