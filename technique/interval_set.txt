/**
 * ある条件を満たす整数を元とする順序付き集合を
 * 極大な連続区間の集合として管理するためのクラス
 * 連続区間は半開区間として保持される
 * 
 * 区間の追加・削除は併合を伴って行われるため
 * 合計で Q 回の処理が要求されている状況において
 * 償却計算量 O(QlogQ) 程度で処理しきれる
 */
template <class T = ll> class IntervalSet {

  static_assert(std::is_integral_v<T> && std::is_signed_v<T>, "T must be a signed integral type");

  static constexpr T SENTINEL = std::numeric_limits<T>::max() / 2;

  using interval = pair<T, T>;
  set<interval> st;
  T element_cnt;

  /**
   * 左端の値が x 以上である区間のうち
   * 最も左端の値が小さいものを指すイテレータを返す
   */
  auto get_lower_bound_iterator(T x) {
    return st.lower_bound({x, x});
  }

public:
  /**
   * 番兵を持った状態でインスタンスが生成される
   */
  IntervalSet() : element_cnt(0) {
    st.insert({SENTINEL, SENTINEL});
    st.insert({-SENTINEL, -SENTINEL});
  }

  /**
   * 条件を満たす整数により構成された連続区間の個数を返す
   */
  ll size() {
    return st.size() - 2;
  }

  /**
   * 連続区間の集合そのものを返す（番兵は消える）
   * コピーして返すため区間の個数 M に対して
   * O(M) の時間・空間計算量が必要となる
   */
  set<interval> all_intervals() {
    set<interval> res = st;
    res.erase(res.begin());
    res.erase(prev(res.end()));
    return res;
  }

  /**
   * 条件を満たす整数の個数を返す
   */
  T number_of_valid_elements() {
    return element_cnt;
  }

  /**
   * 区間 [u,v) に属する全ての整数が
   * 条件を満たす場合に限って true を返す
   * 区間の個数 M に対し時間計算量 O(logM)
   */
  bool is_covered_entirely(T u, T v) {
    assert(-SENTINEL < u);
    assert(u < v);
    assert(v < SENTINEL);

    const auto itr = get_lower_bound_iterator(u);
    if (u == itr->first) {
      return v <= itr->second;
    }

    const auto pitr = prev(itr);
    return v <= pitr->second;
  }

  /**
   * 整数 x が条件を満たす場合に限って true を返す
   * 区間の個数 M に対し時間計算量 O(logM)
   */
  bool is_covered(T x) {
    assert(-SENTINEL < x && x < SENTINEL - 1);
    return is_covered_entirely(x, x + 1);
  }

  /**
   * 区間 [u,v) に属する全ての整数を追加し
   * 実際に増えた整数の個数を返す
   */
  T add_entirely(T u, T v) {
    assert(-SENTINEL < u);
    assert(u < v);
    assert(v < SENTINEL);

    interval added_interval = {u, v};
    T res = 0;

    auto merge_and_subtract_length = [&](const interval &LR) {
      const auto [L, R] = LR;
      chmin(added_interval.first, L);
      chmax(added_interval.second, R);
      res -= R - L;
    };

    auto itr = get_lower_bound_iterator(u);

    if (u < itr->first) {
      const auto pitr = prev(itr);
      if (u <= pitr->second) {
        merge_and_subtract_length(*pitr);
        st.erase(pitr);
      }
    }

    while (itr->first <= v) {
      merge_and_subtract_length(*itr);
      itr = st.erase(itr);
    }

    st.insert(added_interval);
    res += added_interval.second - added_interval.first;
    element_cnt += res;
    return res;
  }

  /**
   * 整数 x を追加し
   * 実際に増えた整数の個数を返す
   */
  T add(T x) {
    assert(-SENTINEL < x && x < SENTINEL - 1);
    return add_entirely(x, x + 1);
  }

  /**
   * 区間 [u,v) に属する全ての整数を削除し
   * 実際に削除された整数の個数を返す
   */
  T erase_entirely(T u, T v) {
    assert(-SENTINEL < u);
    assert(u < v);
    assert(v < SENTINEL);

    T res = 0;
    auto itr = get_lower_bound_iterator(u);

    if (u < itr->first) {
      const auto pitr = prev(itr);
      const auto [L, R] = *pitr;

      if (u < R) {
        st.erase(pitr);

        if (v < R) {
          res += v - u;
          st.insert({L, u});
          st.insert({v, R});
        } else {
          res += R - u;
          st.insert({L, u});
        }
      }
    }

    while (itr->first < v) {
      if (itr->second <= v) {
        res += itr->second - itr->first;
        itr = st.erase(itr);
      } else {
        const auto [L, R] = *itr;
        res += v - L;
        itr = st.erase(itr);
        st.insert({v, R});
        break;
      }
    }

    element_cnt -= res;
    return res;
  }

  /**
   * 整数 x を削除し
   * 実際に削除された整数の個数を返す
   */
  T erase(T x) {
    assert(-SENTINEL < x && x < SENTINEL - 1);
    return erase_entirely(x, x + 1);
  }

  /**
   * x 以上である整数であって
   * 条件を満たさないもののうち最小の値を返す
   * 区間の個数 M に対し時間計算量 O(logM)
   */
  T mex(T x) {
    assert(-SENTINEL < x && x < SENTINEL - 1);
    auto itr = get_lower_bound_iterator(x);
    if (x == itr->first) {
      return itr->second;
    }

    --itr;
    return max(x, itr->second);
  }

  /**
   * 区間 [u,v) に属する全ての整数を含む区間を返す
   * 存在しない場合は番兵のうち負の値を持つ方を返す
   * 区間の個数 M に対し時間計算量 O(logM)
   */
  auto find_entirely_covering_interval(T u, T v) {
    assert(-SENTINEL < u);
    assert(u < v);
    assert(v < SENTINEL);

    auto itr = get_lower_bound_iterator(u);
    if (u < itr->first) {
      --itr;
    }

    if (v <= itr->second) {
      return *itr;
    } else {
      return *st.begin();
    }
  }

  /**
   * 整数 x を含む区間を返す
   * 存在しない場合は番兵のうち負の値を持つ方を返す
   * 区間の個数 M に対し時間計算量 O(logM)
   */
  auto find_covering_interval(T x) {
    assert(-SENTINEL < x && x < SENTINEL - 1);
    return find_entirely_covering_interval(x, x + 1);
  }

  /**
   * x 以上の整数であって
   * 条件を満たすもののうち最小値を返す
   * 存在しない場合は番兵のうち負の値を持つ方を返す
   * 区間の個数 M に対し時間計算量 O(logM)
   */
  T min_valid_value_geq(T x) {
    assert(-SENTINEL < x && x < SENTINEL - 1);

    auto itr = get_lower_bound_iterator(x);
    if (x == itr->first) {
      return x;
    }

    auto pitr = prev(itr);
    if (pitr != st.begin() && x < pitr->second) {
      return x;
    }
    if (itr->first == SENTINEL) {
      return -SENTINEL;
    }
    return itr->first;
  }

  /**
   * x 以下の整数であって
   * 条件を満たすもののうち最大値を返す
   * 存在しない場合は番兵のうち負の値を持つ方を返す
   * 区間の個数 M に対し時間計算量 O(logM)
   */
  T max_valid_value_leq(T x) {
    assert(-SENTINEL < x && x < SENTINEL - 1);

    auto itr = get_lower_bound_iterator(x);
    if (x == itr->first) {
      return x;
    }

    auto pitr = prev(itr);
    if (pitr == st.begin()) {
      return -SENTINEL;
    }
    if (x < pitr->second) {
      return x;
    }
    return pitr->second - 1;
  }
};
