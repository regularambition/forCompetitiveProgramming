/**
 * 10^2以下の素数は25個
 * 10^3以下の素数は168個
 * 10^4以下の素数は1229個
 * 10^5以下の素数は9592個
 * 10^6以下の素数は78498個
 *
 * 直下のSZは1以上10^6以下である
 *******************************
 *******************************
 ********** 全ての **************
 *******************************
 *******************************
 * 整数について素因数分解を事前に求めておく必要あり
 * という想定で書いているため、問題の制約に応じて書き換える
 */
const ll SZ = power(10, 6);
vector<ll> min_factor(SZ + 1, 0);

/**
 * osa_k法を利用して各整数の最小の素因数を求める
 * 基本はエラトステネスの篩と変わらず
 * M以下の整数を対象とする場合について
 * O(MloglogM)の時間計算量となり
 * M <= 10^7くらいまでなら十分間に合う
 */
void osa_k() {
  min_factor[1] = 1;
  for (ll i = 2; i <= SZ; ++i) {
    if (min_factor[i] != 0) {
      continue;
    }

    min_factor[i] = i;
    for (ll j = i * i; j <= SZ; j += i) {
      if (min_factor[j] == 0) {
        min_factor[j] = i;
      }
    }
  }
}

/**
 * 整数Kを素因数分解した結果を返す
 *
 * ただし事前にosa_k()を呼び出して
 * 各整数の最小素因数を求めておく必要あり
 *
 * 時間計算量はO(log(K))
 */
map<ll, ll> prime_factorization(ll K) {
  map<ll, ll> res;
  while (K > 1) {
    ++res[min_factor[K]];
    K /= min_factor[K];
  }
  return res;
}
