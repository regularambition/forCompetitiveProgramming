void tenchi_matrix(vector<string> &mat) {
  int h = mat.size(), w = mat[0].length();

  vector<string> buff(w, string(h, ' '));
  for (int y = 0; y < h; ++y) {
    for (int x = 0; x < w; ++x) {
      buff[x][y] = mat[y][x];
    }
  }

  mat = buff;
}

void reverse_vertically(vector<string> &mat) {
  int h = mat.size();

  for (int y = 0; y < h / 2; ++y) {
    swap(mat[y], mat[h - 1 - y]);
  }
}

void rotate_clockwise(vector<string> &mat) {
  reverse_vertically(mat);
  tenchi_matrix(mat);
}

void rotate_anticlockwise(vector<string> &mat) {
  tenchi_matrix(mat);
  reverse_vertically(mat);
}
