/**
 * セグメント木が以下の要件を満たす場合に使う関数定義
 * 実配列・作用素ともに整数型
 * 実配列が保持する情報：区間最大値
 * 作用素：指定した値とのmaxへと更新
 */
template <class seisu> struct ActionsForSectionMaxSingleMax {
  seisu biop(const seisu &d1, const seisu &d2) const {
    return max(d1, d2);
  }

  seisu id_element() const {
    return -INF64;
  }

  seisu mapping(const seisu &d, const seisu &f) const {
    return max(d, f);
  }
};

template <class seisu> using SegmentTree = SegTree<seisu, seisu, ActionsForSectionMaxSingleMax<seisu>>;
