/**
 * 遅延セグメント木が以下の要件を満たす場合に使う関数定義
 * 実配列・遅延作用素ともに整数型
 * 実配列が保持する情報：区間最大値
 * 遅延作用素：区間に対して一律で特定の値を加算
 */
template <class seisu> struct ActionsForSectionMaxSectionAdd {
  seisu biop(const seisu &d1, const seisu &d2) const {
    return max(d1, d2);
  }

  seisu id_element() const {
    return -INF64;
  }

  seisu mapping(const seisu &d, const seisu &f) const {
    return d + f;
  }

  seisu composition(const seisu &f, const seisu &g) const {
    return g + f;
  }

  seisu id_mapping() const {
    return 0;
  }
};

template <class seisu> using SegmentTree = LazySegTree<seisu, seisu, ActionsForSectionMaxSectionAdd<seisu>>;
